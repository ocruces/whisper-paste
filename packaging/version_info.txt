# Windows VERSIONINFO resource for WhisperPaste.exe, consumed by PyInstaller's
# EXE(version=...) directive.
#
# Not cosmetic. An executable with no version resource scores measurably worse
# with SmartScreen and AV heuristics, and this app already looks keylogger-
# shaped to them (a WH_KEYBOARD_LL hook with suppress=True, plus clipboard
# writes). See README, "Windows SmartScreen and antivirus".
#
# Keep filevers/prodvers and ProductVersion in step with whisper_paste.__version__
# and the version in pyproject.toml — tests/test_packaging.py enforces it.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          u'040904B0',
          [
            StringStruct(u'CompanyName', u'WhisperPaste'),
            StringStruct(u'FileDescription', u'WhisperPaste - voice dictation tray app'),
            StringStruct(u'FileVersion', u'1.0.0.0'),
            StringStruct(u'InternalName', u'WhisperPaste'),
            StringStruct(u'LegalCopyright', u'MIT License. See LICENSE.'),
            StringStruct(u'OriginalFilename', u'WhisperPaste.exe'),
            StringStruct(u'ProductName', u'WhisperPaste'),
            StringStruct(u'ProductVersion', u'1.0.0.0'),
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct(u'Translation', [1033, 1200])])
  ]
)
